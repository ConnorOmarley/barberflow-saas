Step 1 — Initialize Supabase project structure (no Docker required):

Step 2 — Create migration file at supabase/migrations/20260529000001_initial_schema.sql.
The migration must contain ALL of the following in this exact order (no sections may be omitted):

SECTION A — barbershops table:
  CREATE TABLE public.barbershops with columns: id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name TEXT NOT NULL, slug TEXT UNIQUE NOT NULL, timezone TEXT NOT NULL DEFAULT 'America/Sao_Paulo', subscription_status TEXT NOT NULL DEFAULT 'trial', created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW().
  Immediately after: ALTER TABLE public.barbershops ENABLE ROW LEVEL SECURITY;
  RLS policy "owner_own_barbershop" FOR ALL TO authenticated USING (id = (auth.jwt() -> 'app_metadata' ->> 'barbershop_id')::UUID) WITH CHECK (same).

SECTION B — profiles table:
  CREATE TABLE public.profiles with columns: id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, barbershop_id UUID REFERENCES public.barbershops(id) ON DELETE CASCADE (nullable — NULL for new owners before onboarding), role TEXT NOT NULL CHECK (role IN ('owner', 'barber')), full_name TEXT, avatar_url TEXT, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW().
  Immediately after: ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
  Three RLS policies:
    "own_profile_all" FOR ALL TO authenticated USING (id = auth.uid()) WITH CHECK (id = auth.uid())
    "tenant_profiles_select" FOR SELECT TO authenticated USING (barbershop_id = (auth.jwt() -> 'app_metadata' ->> 'barbershop_id')::UUID)
    "auth_admin_read_profiles" FOR SELECT TO supabase_auth_admin USING (true)

SECTION C — clients table:
  CREATE TABLE public.clients with columns: id UUID PRIMARY KEY DEFAULT gen_random_uuid(), barbershop_id UUID NOT NULL REFERENCES public.barbershops(id) ON DELETE CASCADE, full_name TEXT NOT NULL, whatsapp_number TEXT NOT NULL, whatsapp_opt_in BOOLEAN NOT NULL DEFAULT FALSE, opt_in_timestamp TIMESTAMPTZ, opt_in_source TEXT, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW().
  Immediately after: ALTER TABLE public.clients ENABLE ROW LEVEL SECURITY;
  Three RLS policies:
    "tenant_clients_select" FOR SELECT TO authenticated USING (barbershop_id = (auth.jwt() -> 'app_metadata' ->> 'barbershop_id')::UUID)
    "tenant_clients_insert" FOR INSERT TO authenticated WITH CHECK (barbershop_id = (auth.jwt() -> 'app_metadata' ->> 'barbershop_id')::UUID)
    "anon_client_insert" FOR INSERT TO anon WITH CHECK (true)

SECTION D — handle_new_user_from_invite trigger function:
  CREATE OR REPLACE FUNCTION public.handle_new_user_from_invite() RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER.
  Logic: IF NEW.raw_user_meta_data ? 'barbershop_id' THEN insert into public.profiles (id, barbershop_id, role, full_name) values (NEW.id, (NEW.raw_user_meta_data->>'barbershop_id')::UUID, COALESCE(NEW.raw_user_meta_data->>'role','barber'), NEW.raw_user_meta_data->>'full_name') ON CONFLICT (id) DO NOTHING. END IF.
  CREATE TRIGGER on_auth_user_created AFTER INSERT ON auth.users FOR EACH ROW EXECUTE FUNCTION public.handle_new_user_from_invite();

SECTION E — custom_access_token_hook function:
  CREATE OR REPLACE FUNCTION public.custom_access_token_hook(event JSONB) RETURNS JSONB LANGUAGE plpgsql STABLE.
  Logic: SELECT barbershop_id, role INTO user_barbershop_id, user_role FROM public.profiles WHERE id = (event->>'user_id')::UUID.
  Set claims := event->'claims'. Ensure app_metadata object exists (jsonb_typeof check).
  IF user_barbershop_id IS NOT NULL: inject barbershop_id into claims->{app_metadata,barbershop_id} as to_jsonb(user_barbershop_id::TEXT).
  IF user_role IS NOT NULL: inject role into claims->{app_metadata,role} as to_jsonb(user_role).
  Return event with updated claims.
  Permissions after function:
    GRANT USAGE ON SCHEMA public TO supabase_auth_admin;
    GRANT EXECUTE ON FUNCTION public.custom_access_token_hook TO supabase_auth_admin;
    REVOKE EXECUTE ON FUNCTION public.custom_access_token_hook FROM authenticated, anon, public;
    GRANT ALL ON TABLE public.profiles TO supabase_auth_admin;
    REVOKE ALL ON TABLE public.profiles FROM authenticated, anon, public;

CRITICAL constraints for ALL RLS policies:
  - ALWAYS cast to UUID: (auth.jwt() -> 'app_metadata' ->> 'barbershop_id')::UUID — never compare without the cast
  - NEVER use user_metadata (raw_user_meta_data) in any RLS policy
  - NEVER use auth.uid() alone as the tenant isolation check — barbershop_id is always included

Step 3 — Update supabase/config.toml to activate the hook for local dev:
